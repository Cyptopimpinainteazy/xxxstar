10. Security Notes
- timeout risks: Engine strictly enforces `dest_timeout + buffer < source_timeout`. Fails fast if violated.
- finality risks: `FinalityOracle` requires quorum. Single RPC responses are rejected in production mode.
- relayer risks: `Slashing` module tracks `SlashRecord` for censorship or false proofs.
- solver risks: Solver quotes must be signed (interface requires `signature` field).
- VM-specific risks: EVM adapter handles re-entrancy by checking state before claim.

export class AtomicEngine {
    private adapters: Map<VmType, X3VmAdapter> = new Map();
    private ledger: ProofLedger;
    private scoreboard: Scoreboard;
    private timeoutEngine: TimeoutRefundEngine;

    constructor() {
        this.ledger = new ProofLedger();
        this.scoreboard = new Scoreboard();
        this.timeoutEngine = new TimeoutRefundEngine();
    }

    registerAdapter(adapter: X3VmAdapter) {
        this.adapters.set(adapter.vm_type(), adapter);
        console.log(`[Engine] Registered adapter: ${adapter.adapter_name()} (${adapter.vm_type()})`);
    }

    getAdapter(vm: VmType): X3VmAdapter {
        const adapter = this.adapters.get(vm);
        if (!adapter) throw new Error(`Adapter not found for VM: ${vm}`);
        return adapter;
    }

    async submitIntent(intent: AtomicIntent): Promise<IntentId> {
        console.log(`[Engine] Submitting Intent: ${intent.intent_id}`);
        
        // 1. Validate Adapters
        const srcAdapter = this.getAdapter(intent.source_vm);
        const dstAdapter = this.getAdapter(intent.destination_vm);

        if (srcAdapter.readiness_score().total < 50 || dstAdapter.readiness_score().total < 50) {
            throw new Error(`Adapter readiness too low for safe execution.`);
        }

        // 2. Validate Timeout Safety
        const timeoutSafe = this.timeoutEngine.validateTimeoutOrder(
            intent.source_timeout, 
            intent.destination_timeout
        );
        if (!timeoutSafe) {
            throw new Error(`Timeout Safety Check Failed: Destination timeout must be significantly before source timeout.`);
        }

        // 3. Check Chain Health
        const srcHealth = await srcAdapter.chain_health();
        const dstHealth = await dstAdapter.chain_health();
        if (!srcHealth.safe_for_new_intents || !dstHealth.safe_for_new_intents) {
            throw new Error(`Chain health check failed. Pausing new intents.`);
        }

        intent.status = AtomicStatus.RouteQuoted;
        intent.updated_at = Date.now();
        
        await this.ledger.recordIntent(intent);
        return intent.intent_id;
    }

    async executeLock(intent: AtomicIntent): Promise<void> {
        console.log(`[Engine] Executing Source Lock for ${intent.intent_id}`);
        const adapter = this.getAdapter(intent.source_vm);
        
        intent.status = AtomicStatus.SourceLocked;
        const proof = await adapter.lock(intent);
        
        if (!proof.success) {
            intent.status = AtomicStatus.Failed;
            throw new Error(`Source lock failed.`);
        }

        await this.ledger.recordProof(intent.intent_id, "source_lock", proof);
        await this.scoreboard.updateRouteScore(intent.intent_id, "source_lock", 10);
    }

    async executeClaim(intentId: IntentId, preimage: Preimage): Promise<void> {
        const intent = await this.ledger.getIntent(intentId);
        console.log(`[Engine] Executing Claim for ${intentId}`);
        
        const adapter = this.getAdapter(intent.destination_vm);
        const proof = await adapter.claim(intentId, preimage);
        
        if (proof.success) {
            intent.status = AtomicStatus.Claimed;
            await this.ledger.recordProof(intentId, "destination_claim", proof);
            await this.scoreboard.updateRouteScore(intentId, "claim", 10);
        } else {
            throw new Error(`Claim failed.`);
        }
    }
}

export class ProofLedger {
    private intents: Map<IntentId, AtomicIntent> = new Map();
    private proofs: Map<IntentId, Map<string, any>> = new Map();

    async recordIntent(intent: AtomicIntent): Promise<void> {
        this.intents.set(intent.intent_id, intent);
    }

    async getIntent(id: IntentId): Promise<AtomicIntent> {
        const intent = this.intents.get(id);
        if (!intent) throw new Error("Intent not found");
        return intent;
    }

    async recordProof(intentId: IntentId, type: string, proof: any): Promise<void> {
        if (!this.proofs.has(intentId)) this.proofs.set(intentId, new Map());
        this.proofs.get(intentId)!.set(type, proof);
    }
}

export class Scoreboard {
    private routeScores: Map<IntentId, number> = new Map();
    private adapterScores: Map<VmType, number> = new Map();

    updateRouteScore(intentId: IntentId, component: string, points: number) {
        const current = this.routeScores.get(intentId) || 0;
        this.routeScores.set(intentId, current + points);
        console.log(`[Scoreboard] Intent ${intentId} +${points} (${component}): ${current + points}/100`);
    }

    getRouteScore(intentId: IntentId): number {
        return this.routeScores.get(intentId) || 0;
    }
}

export class TimeoutRefundEngine {
    validateTimeoutOrder(sourceTimeout: number, destTimeout: number): boolean {
        // Rule: dest_timeout + buffer < source_timeout
        const buffer = 1000; // 1000 blocks/seconds arbitrary buffer
        return (destTimeout + buffer) < sourceTimeout;
    }
}

export class X3Parser {
    parse(input: string): AtomicIntent {
        // Simplified Recursive Descent Parser for .x3 language
        // Example: atomic swap eth.USDC -> sol.SOL { amount 500 ... }
        
        const cleanInput = input.trim();
        if (!cleanInput.startsWith("atomic swap")) {
            throw new Error("Invalid .x3 syntax: Must start with 'atomic swap'");
        }

        const parts = cleanInput.split("->");
        if (parts.length !== 2) throw new Error("Invalid route syntax");

        const sourcePart = parts[0].trim().replace("atomic swap", "").trim(); // "eth.USDC"
        const destPart = parts[1].split("{")[0].trim(); // "sol.SOL"
        const bodyPart = cleanInput.split("{")[1].replace("}", "").trim(); // "amount 500 ..."

        const [srcChain, srcAsset] = sourcePart.split(".");
        const [dstChain, dstAsset] = destPart.split(".");

        // Parse body (naive key-value)
        let amount = 0n;
        let receiver = "";
        let refundTo = "";
        let mode = RouteMode.DirectHtlc;
        let hashlock = "";
        let srcTimeout = 0;
        let dstTimeout = 0;

        const tokens = bodyPart.split(/\s+/);
        for (let i = 0; i < tokens.length; i++) {
            const token = tokens[i];
            if (token === "amount") amount = BigInt(tokens[++i]);
            if (token === "receiver") receiver = tokens[++i];
            if (token === "refund_to") refundTo = tokens[++i];
            if (token === "mode") mode = tokens[++i] as RouteMode;
            if (token === "hashlock") hashlock = tokens[++i].replace(/["()]/g, ""); // strip sha256()
            if (token === "timeout") {
                if (tokens[i+1] === "source") { srcTimeout = parseInt(tokens[i+2]); i+=2; }
                if (tokens[i+1] === "destination") { dstTimeout = parseInt(tokens[i+2]); i+=2; }
            }
        }

        return {
            intent_id: `intent_${Date.now()}`,
            user: "user_cli",
            source_chain: srcChain,
            destination_chain: dstChain,
            source_vm: this.mapChainToVm(srcChain),
            destination_vm: this.mapChainToVm(dstChain),
            source_asset: srcAsset,
            destination_asset: dstAsset,
            amount_in: amount,
            min_amount_out: 0n,
            receiver: receiver,
            refund_address: refundTo,
            route_mode: mode,
            hashlock: hashlock,
            hash_algorithm: "sha256",
            source_timeout: srcTimeout,
            destination_timeout: dstTimeout,
            max_slippage_bps: 300, // 3%
            deadline: 0,
            finality_requirements: {},
            rpc_quorum_requirement: 3,
            relayer_quorum_requirement: 3,
            status: AtomicStatus.Created,
            created_at: Date.now(),
            updated_at: Date.now()
        };
    }

    private mapChainToVm(chain: string): VmType {
        if (["eth", "1", "polygon", "arb"].includes(chain)) return VmType.Evm;
        if (["sol", "solana"].includes(chain)) return VmType.Svm;
        if (["btc", "bitcoin"].includes(chain)) return VmType.BitcoinScript;
        if (["sui", "aptos"].includes(chain)) return VmType.MoveVm;
        return VmType.X3Vm; // Default fallback
    }
}

export class X3Cli {
    private engine: AtomicEngine;
    private parser: X3Parser;

    constructor() {
        this.engine = new AtomicEngine();
        this.parser = new X3Parser();
        this.registerAdapters();
    }

    private registerAdapters() {
        this.engine.registerAdapter(new EvmAdapter());
        this.engine.registerAdapter(new SvmAdapter());
        this.engine.registerAdapter(new X3VmAdapterImpl());
        this.engine.registerAdapter(new SubstrateAdapter());
        this.engine.registerAdapter(new BitcoinAdapter());
        this.engine.registerAdapter(new MoveAdapter());
        this.engine.registerAdapter(new CosmWasmAdapter());
        this.engine.registerAdapter(new CairoAdapter());
        this.engine.registerAdapter(new PlutusAdapter());
        this.engine.registerAdapter(new TonAdapter());
        this.engine.registerAdapter(new FuelAdapter());
        this.engine.registerAdapter(new NearAdapter());
        this.engine.registerAdapter(new SorobanAdapter());
        this.engine.registerAdapter(new PolkadotPvmAdapter());
        this.engine.registerAdapter(new ZkVmAdapter());
    }

    async runCommand(args: string[]) {
        const command = args[0];
        const input = args.slice(1).join(" ");

        if (command === "compile") {
            try {
                const intent = this.parser.parse(input);
                console.log("\n=== COMPILED INTENT ===");
                console.log(JSON.stringify(intent, null, 2));
                console.log("=======================\n");
                
                // Try to submit to engine to validate adapters
                await this.engine.submitIntent(intent);
                console.log("✅ Intent Valid. Adapters Ready.");
            } catch (e: any) {
                console.error("❌ Compilation/Validation Failed:", e.message);
            }
        } else if (command === "scoreboard") {
            console.log("\n=== ADAPTER SCOREBOARD ===");
            // Iterate all registered adapters and print scores
            // (In a real app, we'd list them from the engine)
            console.log("EVM: 90/100");
            console.log("SVM: 80/100");
            console.log("X3VM: 100/100");
            console.log("Others: 30/100 (Interface Only)");
            console.log("============================\n");
        } else {
            console.log("Usage: x3 compile '<.x3 code>' | x3 scoreboard");
        }
    }
}
