# X3 VM Adapter Scoreboard

Generated by scripts/x3-vm-adapter-proof.sh. Do not edit manually.

Generated: SCOREBOARD_HEADER
echo "  $(date '+%Y-%m-%d %H:%M:%S')" >> "$OUTPUT_SCOREBOARD"

cat >> "$OUTPUT_SCOREBOARD" << 'EOF'

## Adapter Status

Status definitions:
- **COMPLETE**: lock, claim, refund, verify, finality, proof ledger, tests all implemented and passing
- **PARTIAL**: core functions work but missing finality/proof-ledger/tests
- **BROKEN**: has code but stubs/placeholders/no real logic
- **MISSING**: no adapter code exists
- **BLOCKED**: cannot implement without upstream dependency

EOF

# Scan each adapter directory and determine status
declare -A adapter_status
declare -A adapter_evidence
declare -A adapter_missing

# EVM Rust adapter
if [ -f "$PROJECT_ROOT/adapters/evm.rs" ]; then
    placeholder_lines=$(grep -c "Placeholder\|Ok(H256::default())\|Ok(vec!\[\])\|Ok(true)\|Ok(0)\|Ok(())" "$PROJECT_ROOT/adapters/evm.rs" 2>/dev/null || echo 0)
    real_lines=$(grep -c "fn " "$PROJECT_ROOT/adapters/evm.rs" 2>/dev/null || echo 0)
    if [ "$placeholder_lines" -ge 3 ]; then
        adapter_status["EVM (Rust)"]="BROKEN"
        adapter_evidence["EVM (Rust)"]="adapters/evm.rs"
        adapter_missing["EVM (Rust)"]="lock, claim, verify, finality, proof ledger"
    fi
fi

# EVM Solidity HTLC
if [ -f "$PROJECT_ROOT/X3-contracts/evm/contracts/AtlasHTLC.sol" ]; then
    has_lock=$(grep -c "function createHTLC" "$PROJECT_ROOT/X3-contracts/evm/contracts/AtlasHTLC.sol" 2>/dev/null || echo 0)
    has_claim=$(grep -c "function claimHTLC" "$PROJECT_ROOT/X3-contracts/evm/contracts/AtlasHTLC.sol" 2>/dev/null || echo 0)
    has_refund=$(grep -c "function refundHTLC" "$PROJECT_ROOT/X3-contracts/evm/contracts/AtlasHTLC.sol" 2>/dev/null || echo 0)
    has_tests=$(ls "$PROJECT_ROOT/X3-contracts/evm/test/AtlasHTLC.t.sol" 2>/dev/null | wc -l)
    if [ "$has_lock" -ge 1 ] && [ "$has_claim" -ge 1 ] && [ "$has_refund" -ge 1 ]; then
        if [ "$has_tests" -ge 1 ]; then
            if [ -f "$PROJECT_ROOT/X3-contracts/evm/cache/test-failures" ]; then
                adapter_status["EVM (Solidity HTLC)"]="PARTIAL"
                adapter_missing["EVM (Solidity HTLC)"]="tests have failures, no proof ledger, no finality"
            else
                adapter_status["EVM (Solidity HTLC)"]="PARTIAL"
                adapter_missing["EVM (Solidity HTLC)"]="no proof ledger, no finality oracle"
            fi
        fi
    fi
    adapter_evidence["EVM (Solidity HTLC)"]="X3-contracts/evm/contracts/AtlasHTLC.sol"
fi

# Bridge adapters
for vm in Bitcoin Ethereum Solana; do
    lower=$(echo "$vm" | tr '[:upper:]' '[:lower:]')
    file="$PROJECT_ROOT/crates/x3-bridge-adapters/src/${lower}.rs"
    if [ -f "$file" ]; then
        adapter_status["$vm (bridge)"]="PARTIAL"
        adapter_evidence["$vm (bridge)"]="crates/x3-bridge-adapters/src/${lower}.rs"
        adapter_missing["$vm (bridge)"]="lock/claim/refund functions missing, no finality, no proof ledger"
    fi
done

# Missing VMs
for vm in "SVM/Solana" "Substrate" "Cosmos/CosmWasm" "Move/Aptos" "Move/Sui" "Starknet" "TON" "Cardano/eUTXO" "Near" "Fuel" "Polkadot" "Soroban"; do
    adapter_status["$vm"]="MISSING"
    adapter_evidence["$vm"]="No files found"
    adapter_missing["$vm"]="all functions"
done

# Write the table
echo "| VM | Status | Evidence | Missing |" >> "$OUTPUT_SCOREBOARD"
echo "|---|---|---|---|" >> "$OUTPUT_SCOREBOARD"
for vm in "EVM (Rust)" "EVM (Solidity HTLC)" "Bitcoin (bridge)" "Ethereum (bridge)" "Solana (bridge)" "SVM/Solana" "Substrate" "Cosmos/CosmWasm" "Move/Aptos" "Move/Sui" "Starknet" "TON" "Cardano/eUTXO" "Near" "Fuel" "Polkadot" "Soroban"; do
    status="${adapter_status[$vm]:-UNKNOWN}"
    evidence="${adapter_evidence[$vm]:-Unknown}"
    missing="${adapter_missing[$vm]:-Unknown}"
    echo "| $vm | $status | $evidence | $missing |" >> "$OUTPUT_SCOREBOARD"
done

# Compute percentage
total=${#adapter_status[@]}
complete_count=0
for vm in "${!adapter_status[@]}"; do
    if [ "${adapter_status[$vm]}" = "COMPLETE" ]; then
        complete_count=$((complete_count + 1))
    fi
done
pct=$(( complete_count * 100 / (total > 0 ? total : 1) ))

echo "" >> "$OUTPUT_SCOREBOARD"
echo "## Summary" >> "$OUTPUT_SCOREBOARD"
echo "" >> "$OUTPUT_SCOREBOARD"
echo "- **Total VMs tracked**: $total" >> "$OUTPUT_SCOREBOARD"
echo "- **COMPLETE**: $complete_count" >> "$OUTPUT_SCOREBOARD"
echo "- **Overall completion**: ${pct}%" >> "$OUTPUT_SCOREBOARD"
echo "- **Stub/placeholder patterns detected**: $stub_count" >> "$OUTPUT_SCOREBOARD"
echo "" >> "$OUTPUT_SCOREBOARD"

# ---- Generate Proof Ledger ----
cat > "$OUTPUT_PROOF" << 'PROOF_HEADER'
# X3 Proof Ledger

Generated by scripts/x3-vm-adapter-proof.sh. Do not edit manually.

Generated: PROOF_HEADER
echo "  $(date '+%Y-%m-%d %H:%M:%S')" >> "$OUTPUT_PROOF"

cat >> "$OUTPUT_PROOF" << 'EOF'

## Anti-Cheat Scan Results

EOF

echo "- Stub/placeholder patterns found: $stub_count" >> "$OUTPUT_PROOF"
echo "- Adapter tests passed: $pass_count" >> "$OUTPUT_PROOF"
echo "- Adapter tests failed: $fail_count" >> "$OUTPUT_PROOF"
echo "" >> "$OUTPUT_PROOF"

echo "## Adapter Status Details" >> "$OUTPUT_PROOF"
for vm in "${!adapter_status[@]}"; do
    echo "- **$vm**: ${adapter_status[$vm]} — ${adapter_missing[$vm]:-}" >> "$OUTPUT_PROOF"
done

echo ""
echo "=============================================="
echo " Proof Scan Complete"
echo "=============================================="
echo " Scoreboard: $OUTPUT_SCOREBOARD"
echo " Proof Ledger: $OUTPUT_PROOF"
echo " Adapter tests: ${pass_count} passed, ${fail_count} failed"
echo " Stubs found: $stub_count"
echo " Overall completion: ${pct}%"
echo "=============================================="

# Exit with failure if stubs exist (unless in test files)
if [ "$stub_count" -gt 0 ]; then
    echo -e "${YELLOW}[WARN]${NC} $stub_count stub/placeholder patterns detected."
    echo "These should be replaced with real implementations."
fi

if [ "$fail_count" -gt 0 ]; then
    echo -e "${RED}[FAIL]${NC} $fail_count test suites failed."
    exit 1
fi

echo -e "${GREEN}[OK]${NC} All checks passed."
